مثال 1تولید سه محصول (فرز، تراش، سنگ) – بیشینه‌سازی سود

کد پایتون
# مثال 1: تولید سه محصول (ماکزیمم‌سازی سود)
from scipy.optimize import linprog

# max z = 30x1 + 12x2 + 15x3  ->  linprog کمینه می‌کند، پس ضرایب را منفی می‌کنیم
c = [-30, -12, -15]

A_ub = [
    [9, 3, 5],   # فرز   <= 500
    [5, 4, 0],   # تراش  <= 350
    [3, 0, 2],   # سنگ   <= 150
    [0, 0, 1],   # x3    <= 30
]
b_ub = [500, 350, 150, 30]

res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * 3, method="highs")

print("وضعیت:", res.message)
print("x1, x2, x3 =", [round(float(v), 4) for v in res.x])
print("سود بیشینه z =", round(-res.fun, 4))
جواب (خروجی اجرا)
وضعیت: Optimization terminated successfully. (HiGHS Status 7: Optimal)
x1, x2, x3 = [16.6667, 66.6667, 30.0]
سود بیشینه z = 1750.0

مثال 2ماده A و B در یک محصول – کمینه‌سازی هزینه

کد پایتون
# مثال 2: ماده A و B (مینیمم‌سازی هزینه)
from scipy.optimize import linprog

# min z = 4xA + 8xB
c = [4, 8]

# 5xA + 10xB = 150   (محدودیت وزن)
A_eq = [[5, 10]]
b_eq = [150]

# xA <= 20  و  xB >= 14  (به صورت کران متغیرها)
bounds = [(0, 20),     # xA
          (14, None)]  # xB

res = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method="highs")

print("وضعیت:", res.message)
print("xA, xB =", [round(float(v), 4) for v in res.x])
print("هزینه کمینه z =", round(res.fun, 4))
جواب (خروجی اجرا)
وضعیت: Optimization terminated successfully. (HiGHS Status 7: Optimal)
xA, xB = [0.0, 15.0]
هزینه کمینه z = 120.0

مثال 4ساخت ۵ نوع خانه (الف و ب) – بیشینه‌سازی سود

کد پایتون
# مثال 4: ساخت 5 نوع خانه (ماکزیمم‌سازی سود)
from scipy.optimize import linprog

profit = [2000, 3000, 2500, 1700, 2000]
days   = [1+4, 2+7, 2+6, 1+5, 1+3]   # مجموع روزهای کاری A و B برای هر نوع -> [5, 9, 8, 6, 4]

c = [-p for p in profit]
A_ub = [days]
b_ub = [610]

# ---------- بخش الف: بدون حداقل ----------
resA = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * 5, method="highs")
print("=== بخش الف ===")
print("x =", [round(float(v), 4) for v in resA.x])
print("سود بیشینه z =", round(-resA.fun, 4))

# ---------- بخش ب: حداقل 10 خانه از هر نوع ----------
resB = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(10, None)] * 5, method="highs")
print("=== بخش ب ===")
print("x =", [round(float(v), 4) for v in resB.x])
print("سود بیشینه z =", round(-resB.fun, 4))

سود بیشینه z = 305000.0
=== بخش ب ===

مثال 5برنامه تولید سه‌ماهه (عادی و اضافه‌کاری) – کمینه‌سازی هزینه

کد پایتون
# مثال 5: برنامه تولید سه ماهه (وقت عادی و اضافه‌کاری)
from scipy.optimize import linprog

D = [1100, 1200, 1300]   # تقاضای ماه‌ها

# ترتیب متغیرها: [x11, x12, x13, x21, x22, x23]
# (x1i: وقت عادی ماه i ،  x2i: اضافه‌کاری ماه i)
#
# z = 5000*sum(x1i) + 6000*sum(x2i) + 200*sum(x1i + x2i - Di)
#   = (5000+200)*sum(x1i) + (6000+200)*sum(x2i) - 200*sum(Di)
# جمله ثابت -200*sum(D) روی جواب بهینه اثری ندارد؛ در آخر به مقدار هدف اضافه می‌شود.
c = [5200] * 3 + [6200] * 3
const = -200 * sum(D)

A_ub = [
    [1, 1, 1, 0, 0, 0],      # ظرفیت عادی سه ماه:   x11+x12+x13 <= 3000
    [0, 0, 0, 1, 1, 1],      # ظرفیت اضافه‌کاری:    x21+x22+x23 <= 900
    [-1, 0, 0, -1, 0, 0],    # تقاضای ماه 1:        x11+x21 >= 1100
    [0, -1, 0, 0, -1, 0],    # تقاضای ماه 2:        x12+x22 >= 1200
    [0, 0, -1, 0, 0, -1],    # تقاضای ماه 3:        x13+x23 >= 1300
]
b_ub = [3000, 900, -D[0], -D[1], -D[2]]

res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * 6, method="highs")

x = res.x
print("وضعیت:", res.message)
print("وقت عادی  x11,x12,x13 =", [round(float(v), 4) for v in x[:3]])
print("اضافه‌کاری x21,x22,x23 =", [round(float(v), 4) for v in x[3:]])
print("هزینه کمینه z =", round(res.fun + const, 4))
جواب (خروجی اجرا)
وضعیت: Optimization terminated successfully. (HiGHS Status 7: Optimal)
وقت عادی  x11,x12,x13 = [1100.0, 1200.0, 700.0]
اضافه‌کاری x21,x22,x23 = [0.0, 0.0, 600.0]
هزینه کمینه z = 18600000.0

مثال 6شرکت مواد پلاستیکی، سه محصول – بیشینه‌سازی سود

کد پایتون
# مثال 6: شرکت مواد پلاستیکی (سه محصول)
from scipy.optimize import linprog

price = [5, 15, 30]    # قیمت فروش هر بسته 1000 تایی
cost  = [2.5, 4, 8]    # هزینه مستقیم
profit = [p - k for p, k in zip(price, cost)]   # [2.5, 11, 22]

c = [-p for p in profit]

# ساعت مصرفی هر محصول = ساعت ماشین/محوطه + ساعت سرپرست
A_ub = [[0.8 + 0.2, 1.5 + 0.5, 2.5 + 0.5]]   # [1, 2, 3]
b_ub = [200]

res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * 3, method="highs")

print("وضعیت:", res.message)
print("x1, x2, x3 =", [round(float(v), 4) for v in res.x])
print("سود بیشینه z =", round(-res.fun, 4))
جواب (خروجی اجرا)
وضعیت: Optimization terminated successfully. (HiGHS Status 7: Optimal)
x1, x2, x3 = [0.0, 0.0, 66.6667]
سود بیشینه z = 1466.6667

مثال 7سه کارگاه و چهار مشتری – کمینه‌سازی هزینه تولید و حمل

کد پایتون
# مثال 7: سه کارگاه و چهار مشتری (مسئله حمل‌ونقل - مینیمم‌سازی هزینه)
import numpy as np
from scipy.optimize import linprog

demand   = [700, 1500, 400, 500]   # سفارش مشتری 1 تا 4
capacity = [800, 1000, 1700]       # ظرفیت کارگاه 1 تا 3
prod_cost = [45, 50, 60]           # هزینه تولید هر واحد در هر کارگاه

# هزینه حمل: سطر = کارگاه، ستون = مشتری
ship = np.array([
    [8, 11, 6, 9],     # کارگاه 1
    [4, 12, 8, 5],     # کارگاه 2
    [6, 7, 9, 10],     # کارگاه 3
])

m, n = ship.shape
# هزینه کل هر مسیر = حمل + تولید ؛ متغیرها x_ij به صورت سطر به سطر
c = (ship + np.array(prod_cost)[:, None]).flatten()

# محدودیت تقاضا (مساوی): sum_i x_ij = demand_j
A_eq = np.zeros((n, m * n))
for j in range(n):
    for i in range(m):
        A_eq[j, i * n + j] = 1
b_eq = demand

# محدودیت ظرفیت (کوچکتر مساوی): sum_j x_ij <= capacity_i
A_ub = np.zeros((m, m * n))
for i in range(m):
    A_ub[i, i * n:(i + 1) * n] = 1
b_ub = capacity

res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
              bounds=[(0, None)] * (m * n), method="highs")

print("وضعیت:", res.message)
X = res.x.reshape(m, n)
for i in range(m):
    print(f"کارگاه {i+1}:", [round(float(v), 4) for v in X[i]])
print("هزینه کل کمینه z =", round(res.fun, 4))
جواب (خروجی اجرا)
وضعیت: Optimization terminated successfully. (HiGHS Status 7: Optimal)
کارگاه 1: [0.0, 200.0, 400.0, 200.0]
کارگاه 2: [700.0, 0.0, 0.0, 300.0]
کارگاه 3: [0.0, 1300.0, 0.0, 0.0]
هزینه کل کمینه z = 183800.0
